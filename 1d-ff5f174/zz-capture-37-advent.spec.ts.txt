// THROWAWAY (#37 1d captures): not committed. Real server on a December clock (HP_DEV_NOW), real sheet.
import { mkdirSync, writeFileSync } from 'node:fs';
import type { Page } from '@playwright/test';
import { expect, test } from './test.js';
import { hook } from './dev-hook.js';
import { newPlayer, visitPatch } from './players.js';
import { trayButton, traySettled, traysState } from './trays.js';

const OUT = process.env['CAPTURE_OUT'] ?? 'captures';
const TZ = process.env['CAPTURE_TZ'] ?? 'America/Sao_Paulo';
const DEVICE = (process.env['CAPTURE_DEVICE'] ?? 'iphone') as 'iphone' | 'ipad';
const TAG = process.env['CAPTURE_TAG'] ?? 'night';
const LEAD_MS = Number(process.env['CAPTURE_LEAD_MS'] ?? '0');
const SIZES = { iphone: { width: 393, height: 852 }, ipad: { width: 820, height: 1180 } } as const;
const HIDE_DEV = '.dev-status { display: none !important; }';

test(`advent captures ${DEVICE} ${TAG}`, async ({ browser }) => {
  test.setTimeout(15 * 60_000);
  mkdirSync(OUT, { recursive: true });
  Object.assign(test.info().project.use, {
    viewport: SIZES[DEVICE],
    deviceScaleFactor: 2,
    isMobile: true,
    hasTouch: true,
  });
  const name = 'cap' + Math.random().toString(36).slice(2, 7);
  const page = await newPlayer(browser, name, undefined, { timezoneId: TZ });
  if (DEVICE === 'iphone') {
    const cdp = await page.context().newCDPSession(page);
    await cdp.send('Emulation.setSafeAreaInsetsOverride' as never, {
      insets: { top: 59, topMax: 59, bottom: 34, bottomMax: 34, left: 0, leftMax: 0, right: 0, rightMax: 0 },
    } as never);
  }
  await page.context().addInitScript(`(() => {
    const D = Date, lead = ${LEAD_MS};
    class F extends D { constructor(...a) { if (a.length === 0) super(D.now() + lead); else super(...a); } static now() { return D.now() + lead; } }
    window.Date = F;
  })();`);
  await page.goto('/?tier=high');
  await page.addStyleTag({ content: HIDE_DEV });
  const lobby = page.getByTestId('lobby');
  await lobby.getByRole('button', { name: 'Make a patch' }).tap();
  await lobby.getByLabel('Patch name').fill('Snowy Patch');
  await lobby.getByRole('button', { name: 'Make it!' }).tap();
  await visitPatch(lobby);
  await expect(page.locator('#game')).toHaveAttribute('data-ready', 'true', { timeout: 60_000 });
  await page.addStyleTag({ content: HIDE_DEV });
  const hint = page.getByTestId('tray-hint-ok');
  if (await hint.isVisible().catch(() => false)) await hint.tap();
  await page.waitForTimeout(3000);

  const state = (p: Page) => hook<{ days: string[] | null; waiting: boolean }>(p, 'advent');
  await expect.poll(async () => (await state(page))?.waiting, { timeout: 30_000 }).toBe(true);
  const prefix = `${OUT}/${DEVICE}-${TAG}`;
  const shot = async (file: string) => {
    await page.waitForTimeout(900);
    await page.screenshot({ path: `${prefix}-${file}.png` });
    const meta = await page.evaluate(() => {
      const sheet = document.querySelector('[data-testid="advent-sheet"]');
      const r = sheet?.getBoundingClientRect();
      const small = [...(sheet?.querySelectorAll('*') ?? [])]
        .filter((n) => [...n.childNodes].some((c) => c.nodeType === 3 && c.textContent?.trim()))
        .map((n) => ({ t: n.textContent?.trim().slice(0, 24), px: parseFloat(getComputedStyle(n).fontSize) }))
        .filter((x) => x.px < 14);
      const targets = [...(sheet?.querySelectorAll('button') ?? [])].map((b) => {
        const br = b.getBoundingClientRect();
        return { t: b.textContent?.trim().slice(0, 24), w: Math.round(br.width), h: Math.round(br.height) };
      });
      return {
        sheet: r && { top: r.top, bottom: r.bottom, left: r.left, right: r.right },
        inner: { w: innerWidth, h: innerHeight },
        scrolls: sheet ? sheet.scrollHeight - sheet.clientHeight : null,
        smallText: small,
        minTarget: Math.min(...targets.map((t) => Math.min(t.w, t.h))),
        targets: targets.length,
      };
    });
    writeFileSync(`${prefix}-${file}.json`, JSON.stringify({ days: (await state(page))?.days, ...meta }, null, 1));
  };

  // The tray first (the entry with its badge), then the sheet.
  await (await trayButton(page, 'advent-open')).evaluate(() => undefined);
  await traySettled(page, 'heartpatch');
  await page.waitForTimeout(500);
  await page.screenshot({ path: `${prefix}-00-entry.png` });
  await page.getByTestId('advent-open').tap();
  const sheet = page.getByTestId('advent-sheet');
  await expect(sheet).toBeVisible();
  await expect.poll(async () => (await traysState(page))?.open, { timeout: 15_000 }).toBeNull();
  await page.waitForTimeout(1200);
  await expect(sheet.getByTestId('advent-day-5')).toContainText('Today');
  await shot('01-sheet-today-catchup');

  await sheet.getByTestId('advent-day-9').tap();
  await shot('02-locked-tapped');

  await sheet.getByTestId('advent-open-today').tap();
  await expect(sheet.getByTestId('advent-reveal')).toBeVisible();
  await shot('03-reveal-today');
  await sheet.getByTestId('advent-yay').tap();
  await shot('04-today-opened');

  await sheet.getByTestId('advent-day-3').tap();
  await shot('05-catchup-picked');
  await sheet.getByTestId('advent-open-missed').tap();
  await expect(sheet.getByTestId('advent-reveal')).toBeVisible();
  await shot('06-reveal-catchup');
  await sheet.getByTestId('advent-yay').tap();
  await expect(sheet.getByTestId('advent-used')).toBeVisible();
  await shot('07-catchup-used');
});
