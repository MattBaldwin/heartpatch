// THROWAWAY (#190 PR 5 signature moments). Never committed as a spec: it lives in
// pr5-<sha>/zz-capture-190-pr5.spec.ts.txt. Copy it to apps/client/tests/e2e/zz-capture-190-pr5.spec.ts to run.
// The boss fight is boss-stub.ts's three-squishy stand-in (as boss-battle.spec.ts); the app around it is real.
// One run = one device x one time of day. Frames: rest, lean slot 0 / 1 / 2 (the picker), then three rounds:
// a windup round (beat: push-in), its landing round (thump), the winning round (KO + pull-back).
import { mkdirSync, writeFileSync } from 'node:fs';
import { expect, test } from './test.js';
import { stubBossFight } from './boss-stub.js';
import { newPlayer, uniqueName, visitPatch } from './players.js';
import { realTap } from './touch.js';
import { trayButton } from './trays.js';

const OUT = process.env['CAPTURE_OUT'] ?? 'captures';
const TZ = process.env['CAPTURE_TZ'] ?? 'Asia/Dubai';
const TAG = process.env['CAPTURE_TAG'] ?? 'day';
const DEVICE = (process.env['CAPTURE_DEVICE'] ?? 'iphone') as 'iphone' | 'ipad';
const SIZES = { iphone: { width: 393, height: 852 }, ipad: { width: 820, height: 1180 } } as const;
// sig: the signature rounds (+ Flinched!); flinch: rest + the shout round only; win / loss: the end beats.
const PART = process.env['CAPTURE_PART'] ?? 'sig';
const DPR = Number(process.env['CAPTURE_DPR'] ?? '2');

test.use({
  actionTimeout: 90_000,
  launchOptions: {
    executablePath: process.env['PW_CHROMIUM_EXECUTABLE'] ?? '/opt/pw-browsers/chromium',
    args: ['--use-gl=angle', '--use-angle=swiftshader', '--enable-unsafe-swiftshader', '--ignore-gpu-blocklist'],
  },
});

const HIDE_DEV = '.dev-status { display: none !important; }';
const REDUCED = process.env['CAPTURE_REDUCED'] === '1';

interface Now { kind: string | null; at: number; cap: string; live: number }

test(`boss signature captures ${DEVICE} ${TAG}`, async ({ browser }) => {
  test.setTimeout(Number(process.env['CAPTURE_MINUTES'] ?? '25') * 60_000);
  const dir = `${OUT}/${DEVICE}-${TAG}-${PART}`;
  mkdirSync(dir, { recursive: true });
  Object.assign(test.info().project.use, { viewport: SIZES[DEVICE], deviceScaleFactor: DPR, isMobile: true, hasTouch: true });
  const page = await newPlayer(browser, uniqueName('sig'), undefined, { timezoneId: TZ });
  if (REDUCED) await page.emulateMedia({ reducedMotion: 'reduce' });
  // Safe-area insets as on the real devices: iPhone 15 (Dynamic Island) 59/34, iPad (home-bar models) 24/20.
  const INSETS = { iphone: { top: 59, bottom: 34 }, ipad: { top: 24, bottom: 20 } } as const;
  const inset = INSETS[DEVICE];
  const cdp = await page.context().newCDPSession(page);
  await cdp.send('Emulation.setSafeAreaInsetsOverride' as never, {
    insets: { top: inset.top, topMax: inset.top, bottom: inset.bottom, bottomMax: inset.bottom, left: 0, leftMax: 0, right: 0, rightMax: 0 },
  } as never);
  // Capture only: the HUD's sticker hide (CALLOUT_MS 1100, a real timer) never fires, since software
  // GL takes ~1 s a frame and would hide it before the shot; a new word still replaces the old one.
  await page.addInitScript(`(() => {
    const st = window.setTimeout.bind(window);
    window.setTimeout = (fn, ms, ...a) => (ms === 1100 ? 0 : st(fn, ms, ...a));
  })();`);
  // The held battle clock (`?battle-clock=manual`): each beat is reached by advancing it exactly.
  await page.goto('/?tier=high&battle-clock=manual');
  await page.addStyleTag({ content: HIDE_DEV });
  const lobby = page.getByTestId('lobby');
  await lobby.getByRole('button', { name: 'Make a patch' }).tap();
  await lobby.getByLabel('Patch name').fill('Stomp Patch');
  await lobby.getByRole('button', { name: 'Make it!' }).tap();
  await visitPatch(lobby);
  await page.addStyleTag({ content: HIDE_DEV });
  for (let i = 0; i < 2; i++) {
    await realTap(await trayButton(page, 'battle-dev-grant'));
    await expect(page.locator('.battle-entry-note')).toContainText('joined you', { timeout: 30_000 });
  }
  const end = PART === 'win' || PART === 'loss';
  await stubBossFight(
    page,
    end
      ? PART === 'win'
        ? { warmth: 0, befriend: ['not-yet', 'joined'] }
        : { outcome: 'lost', triesLeftToday: 1 }
      : PART === 'flinch'
        ? { openRounds: 1, tricks: ['shout'], signatures: true }
        : { openRounds: 5, tricks: ['windup', 'land', 'shout', 'assist', 'drift'], signatures: true },
  );
  await realTap(await trayButton(page, 'battle-dev-fight'));
  await expect(page.getByTestId('boss-strip')).toBeVisible({ timeout: 60_000 });
  await page.addStyleTag({ content: HIDE_DEV });

  /** A page.evaluate that survives the odd navigation-free context reset under SwiftShader. */
  const retry = async <T,>(fn: () => Promise<T>): Promise<T> => {
    for (let i = 0; ; i++) {
      try {
        return await fn();
      } catch (e) {
        if (i >= 5 || !/Execution context was destroyed|Target closed|navigat/i.test(String(e))) throw e;
        await page.waitForTimeout(1000);
      }
    }
  };
  const advance = (ms: number) =>
    retry(() => page.evaluate((m) => (window as any).__heartpatch.battleDev().advance(m), ms));
  const now = (): Promise<Now> =>
    retry(() => page.evaluate(() => {
      const s = (window as any).__heartpatch?.battle?.()?.scene;
      return {
        kind: s?.playing?.kind ?? null,
        at: s?.playing?.at ?? 0,
        cap: (document.querySelector('[data-testid=battle-caption]')?.textContent ?? '').trim(),
        live: s?.effects?.live ?? 0,
      };
    }));
  /** Advances the held clock in `step` ms until `want` holds; tracks the most particles alive on the way. */
  let peak = 0;
  const until = async (label: string, want: (n: Now) => boolean, step = 20, max = 60_000) => {
    for (let t = 0; t <= max; t += step) {
      const n = await now();
      peak = Math.max(peak, n.live);
      if (want(n)) return n;
      await advance(step);
    }
    throw new Error(`never saw ${label}`);
  };
  /** In one jump (stickers run on real timers, so a slow walk would outlast them): to `ms` into the step now playing. */
  const jump = async (ms: number) => {
    const n = await now();
    // Most of the way in one go, then the last 160 ms in 20 ms frames, so the camera eases as at 60 fps.
    const left = Math.ceil(ms - n.at);
    if (left > 160) await advance(left - 160);
    for (let t = Math.min(160, Math.max(0, left)); t > 0; t -= 20) await advance(Math.min(20, t));
  };
  const rect = (sel: string) =>
    page.evaluate((s) => {
      const e = document.querySelector(s) as HTMLElement | null;
      if (!e || e.hidden) return null;
      const r = e.getBoundingClientRect();
      return { top: r.top / innerHeight, bottom: r.bottom / innerHeight, left: r.left, right: r.right, text: e.textContent };
    }, sel);
  const shot = async (file: string) => {
    await page.waitForTimeout(600);
    const path = `${dir}/${DEVICE}-${file}-${TAG}.png`;
    await page.screenshot({ path });
    const meta = {
      file: path.split('/').pop(),
      device: DEVICE,
      tz: TZ,
      reduced: REDUCED,
      inner: await page.evaluate(() => ({ w: window.innerWidth, h: window.innerHeight, dpr: devicePixelRatio })),
      safeArea: await page.evaluate(() => {
        const d = document.createElement('div');
        d.style.cssText = 'position:fixed;top:0;left:0;padding-top:env(safe-area-inset-top);padding-bottom:env(safe-area-inset-bottom);visibility:hidden';
        document.body.appendChild(d);
        const c = getComputedStyle(d);
        const r = { top: parseFloat(c.paddingTop), bottom: parseFloat(c.paddingBottom) };
        d.remove();
        return r;
      }),
      nameBar: await rect('[data-testid=boss-strip] .boss-strip-name, [data-testid=boss-strip]'),
      endCard: await rect('[data-testid=battle-result]'),
      battle: await page.evaluate(() => {
        const s = (window as any).__heartpatch?.battle?.()?.scene;
        return s ? { effects: s.effects, drawCalls: s.drawCalls, playing: s.playing, framing: s.framing, camera: s.camera } : null;
      }),
      peakLiveThisRound: peak,
      caption: (await now()).cap,
      sticker: await rect('[data-testid=boss-callout]'),
      strip: await rect('[data-testid=boss-strip]'),
      handle: await rect('[data-testid=boss-handle]'),
      friendCallout: await rect('.boss-callout-small'),
      captionPill: await rect('[data-testid=battle-caption]'),
    };
    writeFileSync(path.replace(/\.png$/, '.json'), JSON.stringify(meta, null, 1));
    console.log(`shot ${meta.file} at=${Math.round((meta.battle as any)?.playing?.at ?? -1)} live=${(meta.battle as any)?.effects?.live} draws=${(meta.battle as any)?.drawCalls} peak=${peak} sticker=${JSON.stringify(meta.sticker?.text ?? null)}`);
  };

  /** Every mesh the battle scene drew last frame, projected to CSS px (finds what a stray shape is). */
  const meshDump = async (file: string) => {
    const out = await retry(() => page.evaluate(async () => {
      const deps = performance.getEntriesByType('resource').map((e) => e.name).find((n) => n.includes('/.vite/deps/'));
      if (!deps) return { err: 'no vite deps url' };
      const v = new URL(deps).searchParams.get('v');
      const { EngineStore } = await import(`/node_modules/.vite/deps/@babylonjs_core_Engines_engineStore.js${v ? `?v=${v}` : ''}`);
      const W = innerWidth, H = innerHeight;
      const res: unknown[] = [];
      for (const engine of EngineStore.Instances) for (const scene of engine.scenes) {
        if (!scene.activeCamera) continue;
        const m = scene.getTransformMatrix().m;
        const proj = (x: number, y: number, z: number) => {
          const w = x * m[3] + y * m[7] + z * m[11] + m[15];
          const px = (x * m[0] + y * m[4] + z * m[8] + m[12]) / w, py = (x * m[1] + y * m[5] + z * m[9] + m[13]) / w;
          return { x: Math.round(((px + 1) / 2) * W), y: Math.round(((1 - py) / 2) * H) };
        };
        const col = (c: any) => (c ? [c.r, c.g, c.b].map((n: number) => Math.round(n * 255)) : null);
        const meshes = scene.getActiveMeshes().data.slice(0, scene.getActiveMeshes().length);
        for (const mesh of meshes) {
          const bs = mesh.getBoundingInfo().boundingSphere;
          const c = bs.centerWorld;
          const mat = mesh.material;
          const row: any = {
            name: mesh.name, cls: mesh.getClassName(), parent: mesh.parent?.name ?? null,
            mat: mat?.name ?? null, diffuse: col(mat?.diffuseColor ?? mat?.albedoColor), emissive: col(mat?.emissiveColor),
            alpha: mat?.alpha, vis: mesh.visibility, at: proj(c.x, c.y, c.z), r: Math.round(bs.radiusWorld * 1000) / 1000,
            world: [c.x, c.y, c.z].map((n: number) => Math.round(n * 100) / 100),
          };
          const n = mesh.thinInstanceCount ?? 0;
          if (n > 0) {
            row.thin = n;
            const data = mesh._thinInstanceDataStorage?.matrixData as Float32Array | undefined;
            const wm = mesh.getWorldMatrix().m;
            if (data && n <= 64) row.thinAt = Array.from({ length: n }, (_, i) => {
              const tx = data[i * 16 + 12], ty = data[i * 16 + 13], tz = data[i * 16 + 14];
              const sx = Math.hypot(data[i * 16], data[i * 16 + 1], data[i * 16 + 2]);
              const x = tx * wm[0] + ty * wm[4] + tz * wm[8] + wm[12];
              const y = tx * wm[1] + ty * wm[5] + tz * wm[9] + wm[13];
              const z = tx * wm[2] + ty * wm[6] + tz * wm[10] + wm[14];
              return { ...proj(x, y, z), s: Math.round(sx * 1000) / 1000 };
            });
          }
          res.push(row);
        }
      }
      return { W, H, meshes: res };
    }));
    writeFileSync(`${dir}/${DEVICE}-${file}-${TAG}.meshes.json`, JSON.stringify(out, null, 1));
  };
  const picker = page.getByTestId('boss-picker');
  const toPicker = async () => {
    for (let i = 0; i < 400; i++) {
      if ((await picker.isVisible()) && (await page.getByTestId('boss-pick-header').isVisible())) return;
      await advance(200);
    }
    throw new Error('no picker');
  };
  const go = async () => {
    await toPicker();
    for (let i = 0; i < 3; i++) {
      await expect(page.getByTestId('boss-pick-header')).toBeVisible();
      await realTap(picker.getByTestId('boss-move').first());
      await advance(600);
    }
    for (let i = 0; i < 50 && !((await picker.textContent()) ?? '').includes("Everyone's ready!"); i++) await advance(200);
    await expect(picker).toContainText("Everyone's ready!");
    await realTap(page.getByTestId('boss-go'));
    peak = 0;
  };

  // Rest: past the entrance, the picker up.
  await toPicker();
  // The entrance (2.6 s of battle time) covers the picker until it ends.
  for (let i = 0; i < 100; i++) {
    if ((await retry(() => page.evaluate(() => (window as any).__heartpatch?.battle?.()?.entrance ?? null))) === null) break;
    await advance(200);
  }
  await advance(1500);
  await shot('00-rest');
  if (PART === 'flinch') {
    await go();
    await until('shout', (n) => n.kind === 'move' && /Full Name Shout/.test(n.cap));
    await jump(260);
    await shot('02-shout');
    await until('flinch', (n) => n.kind === 'effect' && /flinch/i.test(n.cap));
    await jump(250);
    await shot('02b-flinched');
    await meshDump('02b-flinched');
    // A step on, to see whether the shape follows the flinch beat.
    await jump(600);
    await shot('02c-flinched-later');
    await meshDump('02c-flinched-later');
    return;
  }
  if (end) {
    await go();
    if (PART === 'win') {
      await until('tuckered', (n) => n.kind === 'tuckered');
      await jump(700);
      await shot('05-tuckered');
    } else {
      await until('team tuckered', (n) => /Your team is all tuckered out/.test(n.cap));
      await jump(500);
      await shot('06-loss-twirl');
      await jump(1200);
      await shot('06b-loss-twirl-late');
    }
    const card = page.getByTestId('battle-result');
    for (let i = 0; i < 300 && !(await card.isVisible()); i++) await advance(200);
    await expect(card).toBeVisible();
    await advance(800);
    await shot(PART === 'win' ? '07-ask-0of4' : '06c-loss-card');
    if (PART === 'win') {
      await realTap(card.getByTestId('boss-ask'));
      await expect(card.getByTestId('boss-befriend-ask')).toContainText('A little warmer');
      await advance(400);
      await shot('07b-not-yet');
      await realTap(card.getByTestId('boss-ask'));
      await expect(page.locator('.befriend-dim')).toHaveCount(1);
      const dimAt = () => page.evaluate(() => {
        const d = document.querySelector('.befriend-dim') as HTMLElement | null;
        const a = d?.getAnimations()[0];
        return d ? { opacity: getComputedStyle(d).opacity, animMs: a ? Math.round(Number(a.currentTime)) : null, state: a?.playState ?? null } : null;
      });
      const dimSeen = await dimAt();
      await advance(700);
      await page.waitForTimeout(600);
      const dimBefore = await dimAt();
      console.log(`dim: on appear ${JSON.stringify(dimSeen)}; at the old shot time ${JSON.stringify(dimBefore)}`);
      writeFileSync(`${dir}/dim-check.json`, JSON.stringify({ onAppear: dimSeen, atOldShotTime: dimBefore }, null, 1));
      await shot('08-joining-as-before');
      // Capture only: hold the beat's CSS animations at their peak (25 % of 1500 ms), as a real-time 60 fps phone shows it.
      await page.evaluate(() => {
        for (const a of document.querySelector('.befriend-beat')?.getAnimations({ subtree: true }) ?? []) {
          a.pause();
          a.currentTime = 375;
        }
      });
      await shot('08-joining');
      await expect(card.getByTestId('boss-befriend-joined')).toBeVisible({ timeout: 30_000 });
      await advance(500);
      await shot('08b-joined');
    }
    return;
  }
  // Round 1: the count, the pile grown ("1…" on the first stomp).
  await go();
  await until('windup', (n) => n.kind === 'effect' && /winding up/i.test(n.cap));
  await jump(520);
  await shot('01-leaf-pile');
  // Round 2: it lands ("3!"), and the leaves burst on the first hit.
  await go();
  await until('landing move', (n) => n.kind === 'move' && /Count to Three/.test(n.cap));
  await jump(200);
  await shot('01a-three');
  await until('landing hit', (n) => n.kind === 'hit');
  await jump(380);
  await shot('01b-leaf-burst');
  // Round 3: the shout: hair flare and rings.
  await go();
  await until('shout', (n) => n.kind === 'move' && /Full Name Shout/.test(n.cap));
  await jump(260);
  await shot('02-shout');
  await until('flinch', (n) => n.kind === 'effect' && /flinch/i.test(n.cap));
  await jump(250);
  await shot('02b-flinched');
  // Round 4: the call (shrug, "Backup!"), then the friend's fire flash.
  await go();
  await until('call', (n) => n.kind === 'effect' && /called for backup/i.test(n.cap));
  await jump(300);
  await shot('03-backup-call');
  await until('backup move', (n) => n.kind === 'move' && /^Backup!/.test(n.cap));
  await jump(110);
  await shot('03b-backup-flash');
  // Round 5: the drift: skid arcs and the 2° lean.
  await go();
  await until('drift', (n) => n.kind === 'move' && /Drift Spin/.test(n.cap));
  await jump(440);
  await shot('04-drift');
  // Its spark trail at a friend's post (the hit beat).
  await until('drift hit', (n) => n.kind === 'hit');
  await jump(200);
  await shot('04b-drift-hit');
});
